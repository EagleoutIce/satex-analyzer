\begin{algorithm}\caption{Binary search}\label{alg:b}
\begin{algorithmic}[1]\Require sorted $A$, key $k$
\Procedure{Search}{$A,k$}\State $l\gets 1$, $h\gets n$
\While{$l\le h$}\State $m\gets\lfloor(l+h)/2\rfloor$
\If{$A[m]=k$}\State \Return $m$\ElsIf{$A[m]<k$}\State $l\gets m+1$\Else\State $h\gets m-1$\EndIf
\EndWhile\State \Return $-1$\EndProcedure\For{$i=1$ \textbf{to} $n$}\State work\EndFor\Repeat\State x\Until{done}
\end{algorithmic}\end{algorithm}
