\begin{algorithm}[H]\caption{Euclid}\label{alg:e}
\KwIn{$a,b\in\mathbb N$}\KwOut{$\gcd(a,b)$}
\While{$b\neq0$}{$t\leftarrow b$\; $b\leftarrow a\bmod b$\; $a\leftarrow t$\;}
\Return{$a$}\end{algorithm}

\end{algorithm}
